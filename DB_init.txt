
CREATE TABLE RISTORANTEREGISTRATO(
IDRISTORANTE INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
NOME VARCHAR(100),
USERNAME VARCHAR(30),
PASSWORD VARCHAR(30),
INDIRIZZO VARCHAR(100),
TELEFONO VARCHAR(15),
EMAIL VARCHAR(50)
);

CREATE TABLE FATTORINO(
IDFATTORINO INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
NOME VARCHAR(100),
USERNAME VARCHAR(30),
PASSWORD VARCHAR(30),
STATO ENUM('DISPONIBILE', 'OCCUPATO')
);

CREATE TABLE PESCHERIA(
IDPESCHERIA INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
NOME VARCHAR(100),
INDIRIZZO VARCHAR(100),
EMAIL VARCHAR(50),
USERNAME VARCHAR(30),
PASSWORD VARCHAR(30)
);

CREATE TABLE PRODOTTO(
IDPRODOTTO INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
CATEGORIA VARCHAR(50),
TIPOLOGIA VARCHAR(50),
DESCRIZIONE VARCHAR(255),
PREZZO FLOAT,
CODICEPAESE VARCHAR(10),
IDPESCHERIA INT,
FOREIGN KEY (IDPESCHERIA) REFERENCES PESCHERIA(IDPESCHERIA)
);

CREATE TABLE ORDINE(
IDORDINE INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
IDRISTORANTE INT,
IDPESCHERIA INT,
IDPRODOTTO INT,
DATA DATE,
QTA FLOAT,
QTAAGGIORNATA FLOAT,
PREZZO FLOAT,
STATO ENUM('IN_TRATTATIVA', 'CONFERMATO', 'ASSEGNATO', 'CONSEGNATO'),
FOREIGN KEY (IDRISTORANTE) REFERENCES RISTORANTEREGISTRATO(IDRISTORANTE),
FOREIGN KEY (IDPESCHERIA) REFERENCES PESCHERIA(IDPESCHERIA),
FOREIGN KEY (IDPRODOTTO) REFERENCES PRODOTTO(IDPRODOTTO)
);

CREATE TABLE CONSEGNA(
IDCONSEGNA INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
IDFATTORINO INT,
DATA DATE,
FOREIGN KEY (IDFATTORINO) REFERENCES FATTORINO(IDFATTORINO)
);
